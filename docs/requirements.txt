# System Requirements

## Neovim
brew install neovim   # must be >= 0.12.0

## Core Tools
brew install ripgrep fd git curl tar unzip

## Fonts (pick one Nerd Font)
brew install --cask font-jetbrains-mono-nerd-font
brew install --cask font-space-mono-nerd-font

## Runtimes (for LSP servers via Mason)
brew install node npm       # JS/TS/YAML/HTML/CSS LSPs
brew install go             # gopls
brew install python         # ruff, debugpy
brew install rust           # rust-analyzer, tree-sitter-cli

## tree-sitter CLI (required by nvim-treesitter main branch)
cargo install tree-sitter-cli --locked

## Formatters (or install via :Mason inside Neovim)
brew install stylua prettier shfmt

## Git tools
brew install lazygit gh

## Image rendering (optional, for snacks.image)
brew install imagemagick ghostscript
